package tcp

import "gvisor.dev/gvisor/pkg/waiter"

// WaitHangup observes the same completed TCP exchange as Wait, including a
// TIME_WAIT transition that happened before this call. Registering before the
// state check prevents a lost notification between inspection and sleep.
//
// Like Wait's already-registered EventHUp observer, this does not wait for the
// TIME_WAIT timer or join the stack's dispatcher. Those remain stack-owned and
// must be retired separately. It neither closes nor aborts the endpoint.
func (e *Endpoint) WaitHangup() {
	waitEntry, notifyCh := waiter.NewChannelEntry(waiter.EventHUp)
	e.waiterQueue.EventRegister(&waitEntry)
	defer e.waiterQueue.EventUnregister(&waitEntry)
	switch e.EndpointState() {
	case StateClose, StateError, StateTimeWait:
		return
	}
	<-notifyCh
}
